{{- $page := .Page -}} {{- $images := sort ($page.Resources.ByType "image") "Name" "asc" -}} {{- $items := slice -}} {{- range $images -}} {{- $taken := "" -}} {{- with .Params.taken -}} {{- $taken = time.Format "2006-01-02T15:04:05" . -}} {{- else -}} {{- with .Exif -}} {{- with .Date -}} {{- $taken = time.Format "2006-01-02T15:04:05" . -}} {{- end -}} {{- end -}} {{- end -}} {{- $items = $items | append (dict "image" . "taken" $taken) -}} {{- end -}}
{{- range sort $items "taken" "desc" }} {{- $image := .image -}} {{- $stem := replaceRE `\.[^.]+$` "" $image.Name -}} {{- $captionName := printf "%s.txt" $stem -}} {{- $caption := strings.Trim ($image.Params.caption | default "") " \t\r\n" -}} {{- if not $caption -}} {{- with $page.Resources.GetMatch $captionName -}} {{- $caption = strings.Trim .Content " \t\r\n" -}} {{- end -}} {{- end -}} {{- $alt := $image.Params.alt | default (humanize $stem) -}} {{- end }}
